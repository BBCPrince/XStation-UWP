cmake_minimum_required(VERSION 3.24)
project(XStation LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "WindowsStore")
  message(FATAL_ERROR "This target is a UWP app; configure with CMAKE_SYSTEM_NAME=WindowsStore")
endif()
file(GLOB assets CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/Assets/*.png")
foreach(file IN LISTS assets)
  set_source_files_properties("${file}" PROPERTIES VS_DEPLOYMENT_CONTENT 1 VS_DEPLOYMENT_LOCATION "Assets")
endforeach()
set(XSTATION_PKG_KEYS_DIR "" CACHE PATH "Local folder containing pkg_keys.h (not distributed)")
if(NOT EXISTS "${XSTATION_PKG_KEYS_DIR}/pkg_keys.h")
  message(FATAL_ERROR "Provide -DXSTATION_PKG_KEYS_DIR=<local folder containing pkg_keys.h>. See docs/BUILD.md; key material is not included in this repository.")
endif()
add_executable(XStation WIN32 native_cache_bridge.cpp game_app.cpp dashboard.cpp transfer.cpp usb_files.cpp pkg_extract.cpp game_prepare.cpp tlou_assets.cpp third_party/miniz/miniz_tinfl.c Package.appxmanifest ${assets})
target_compile_features(XStation PRIVATE cxx_std_20)
target_include_directories(XStation PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${XSTATION_PKG_KEYS_DIR}")
target_compile_definitions(XStation PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
option(XSTATION_PUBLIC "Build the public host with fake-PKG-only imports and development switches disabled" ON)
if(XSTATION_PUBLIC)
  target_compile_definitions(XStation PRIVATE XSTATION_PUBLIC)
endif()
target_compile_options(XStation PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W3 /permissive- /utf-8 /EHsc /bigobj>)
target_link_libraries(XStation PRIVATE windowsapp runtimeobject dxgi bcrypt ws2_32)
# As the TLOU host: guest entry and guest pthreads keep explicitly committed stacks.
target_link_options(XStation PRIVATE /HIGHENTROPYVA:NO /STACK:16777216,65536)
set_target_properties(XStation PROPERTIES VS_WINDOWS_TARGET_PLATFORM_MIN_VERSION "10.0.19041.0")

set(XSTATION_RUNTIME_STAGE "" CACHE PATH "Runtime content generated by build/stage-runtimes.ps1")
if(XSTATION_RUNTIME_STAGE)
  file(GLOB_RECURSE runtime_content CONFIGURE_DEPENDS "${XSTATION_RUNTIME_STAGE}/*")
  list(FILTER runtime_content EXCLUDE REGEX "\\.before-")
  foreach(file IN LISTS runtime_content)
    file(RELATIVE_PATH relative "${XSTATION_RUNTIME_STAGE}" "${file}")
    get_filename_component(destination "${relative}" DIRECTORY)
    file(TO_NATIVE_PATH "${destination}" destination)
    set_source_files_properties("${file}" PROPERTIES VS_TOOL_OVERRIDE "None" VS_DEPLOYMENT_CONTENT 1 VS_DEPLOYMENT_LOCATION "${destination}")
  endforeach()
  target_sources(XStation PRIVATE ${runtime_content})
endif()
